java
@Component
@Slf4j
public class JwtTokenProvider {
    @Value("${jwt.secret}")
    private String jwtSecret;

    @Value("${jwt.expiration}")
    private long jwtExpiration;

    public String generateToken(UserDetails userDetails) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("roles", userDetails.getAuthorities().stream()
                .map(GrantedAuthority::getAuthority)
                .collect(Collectors.toList()));

        return Jwts.builder()
                .setClaims(claims)
                .setSubject(userDetails.getUsername())
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + jwtExpiration))
                .signWith(Keys.hmacShaKeyFor(jwtSecret.getBytes()))
                .compact();
    }

    public String getUsernameFromToken(String token) {
        return Jwts.parserBuilder()
                .setSigningKey(Keys.hmacShaKeyFor(jwtSecret.getBytes()))
                .build()
                .parseClaimsJws(token)
                .getBody()
                .getSubject();
    }

    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder()
                    .setSigningKey(Keys.hmacShaKeyFor(jwtSecret.getBytes()))
                    .build()
                    .parseClaimsJws(token);
            return true;
        } catch (JwtException | IllegalArgumentException e) {
            log.error("Invalid JWT token: {}", e.getMessage());
            return false;
        }
    }
}

     * Process XP gain when task completed
     */
    public void awardXP(UUID userId, Integer xpAmount) {
        UserStats stats = statsRepository.findByUserId(userId)
                .orElseThrow(() -> new RuntimeException("Stats not found"));

        // Update stats
        stats.setTotalXp(stats.getTotalXp() + xpAmount);
        stats.setTasksCompleted(stats.getTasksCompleted() + 1);

        // Level up logic: XP needed = level * 100
        int newLevel = calculateLevel(stats.getTotalXp());
        if (newLevel > stats.getLevel()) {
            stats.setLevel(newLevel);
            log.info("User {} leveled up to {}", userId, newLevel);
        }

        statsRepository.save(stats);

        // Send XP event for async processing
        XPUpdateEvent event = XPUpdateEvent.builder()
                .userId(userId)
                .xpGained(xpAmount)
                .newTotal(stats.getTotalXp())
                .newLevel(stats.getLevel())
                .timestamp(Instant.now())
                .build();

        kafkaTemplate.send(XP_EVENT_TOPIC, event);

        // Cache new stats in Redis
        String cacheKey = "user_stats:" + userId;
        redisTemplate.opsForValue().set(cacheKey, stats, 1, TimeUnit.HOURS);

        // Check for badge unlocks
        checkAndAwardBadges(userId, stats);
    }

    /**
     * Maintain daily streaks
     */
    @Scheduled(cron = "0 0 0 * * *") // Midnight daily
    public void updateStreaks() {
        List<UserStats> allStats = statsRepository.findAll();

        for (UserStats stats : allStats) {
            LocalDate today = LocalDate.now();
            LocalDate lastActivity = stats.getLastActivityDate();

            if (lastActivity != null && lastActivity.equals(today.minusDays(1))) {
                // Consecutive day
                stats.setCurrentStreak(stats.getCurrentStreak() + 1);
                if (stats.getCurrentStreak() > stats.getLongestStreak()) {
                    stats.setLongestStreak(stats.getCurrentStreak());
                }
            } else if (lastActivity == null || !lastActivity.equals(today)) {
                // Streak broken
                stats.setCurrentStreak(0);
            }

            statsRepository.save(stats);
        }
    }

    private int calculateLevel(int totalXp) {
        return (int) Math.floor((Math.sqrt(100 * totalXp + 25) + 5) / 100);
    }

    private void checkAndAwardBadges(UUID userId, UserStats stats) {
        // Badge criteria examples
        Map<String, Boolean> badgeConditions = Map.of(
            "FIRST_TASK", stats.getTasksCompleted() >= 1,
            "TASK_MASTER", stats.getTasksCompleted() >= 50,
            "STREAK_3", stats.getCurrentStreak() >= 3,
            "STREAK_7", stats.getCurrentStreak() >= 7,
            "LEVEL_5", stats.getLevel() >= 5,
            "LEVEL_10", stats.getLevel() >= 10,
            "XP_1000", stats.getTotalXp() >= 1000
        );

        badgeConditions.forEach((badgeName, condition) -> {
            if (condition && !userBadgeRepository.hasBadge(userId, badgeName)) {
                Badge badge = badgeRepository.findByName(badgeName);
                if (badge != null) {
                    UserBadge userBadge = UserBadge.builder()
                            .userId(userId)
                            .badge(badge)
                            .earnedAt(LocalDateTime.now())
                            .build();
                    userBadgeRepository.save(userBadge);
                    log.info("Badge awarded: {} to user {}", badgeName, userId);
                }
            }
        });
    }
}

@Service
@RequiredArgsConstructor
public class DashboardService {
    private final UserStatsRepository statsRepository;
    private final TaskRepository taskRepository;
    private final RedisTemplate<String, Object> redisTemplate;

    public DashboardDTO getUserDashboard(UUID userId) {
        // Try cache first
        String cacheKey = "dashboard:" + userId;
        DashboardDTO cached = (DashboardDTO) redisTemplate.opsForValue().get(cacheKey);
        if (cached != null) return cached;

        UserStats stats = statsRepository.findByUserId(userId)
                .orElseThrow(() -> new RuntimeException("Stats not found"));

        // Task completion statistics
        LocalDate weekStart = LocalDate.now().with(DayOfWeek.MONDAY);
        List<TaskCompletionProjection> weeklyCompletions =
                taskRepository.getWeeklyCompletions(userId, weekStart);

        // Category breakdown
        List<CategoryBreakdown> categoryBreakdown =
                taskRepository.getCategoryBreakdown(userId);

        DashboardDTO dashboard = DashboardDTO.builder()
                .totalXp(stats.getTotalXp())
                .level(stats.getLevel())
                .xpToNextLevel(getXpForNextLevel(stats.getLevel()) - stats.getTotalXp())
                .currentStreak(stats.getCurrentStreak())
                .longestStreak(stats.getLongestStreak())
                .tasksCompleted(stats.getTasksCompleted())
                .completionRate(calculateCompletionRate(userId))
                .weeklyCompletions(weeklyCompletions)
                .categoryBreakdown(categoryBreakdown)
                .build();

        // Cache for 5 minutes
        redisTemplate.opsForValue().set(cacheKey, dashboard, 5, TimeUnit.MINUTES);

        return dashboard;
    }

    private double calculateCompletionRate(UUID userId) {
        long total = taskRepository.countByUserId(userId);
        long completed = taskRepository.countByUserIdAndCompletedTrue(userId);
        return total > 0 ? (double) completed / total * 100 : 0;
    }

    private int getXpForNextLevel(int level) {
        return (int) Math.pow(level + 1, 2) * 100;
    }
}

@Service
@RequiredArgsConstructor
@Slf4j
public class OpenAIService {
    private final TaskRepository taskRepository;
    private final UserStatsRepository statsRepository;

    @Value("${openai.api-key}")
    private String apiKey;

    /**
     * Generate weekly productivity report using OpenAI
     */
    public AIReportDTO generateWeeklyReport(UUID userId) {
        // Gather user data
        List<Task> weeklyTasks = taskRepository.findTasksForWeek(userId,
                LocalDate.now().with(DayOfWeek.MONDAY));
        UserStats stats = statsRepository.findByUserId(userId).get();

        // Build prompt
        String prompt = buildPrompt(weeklyTasks, stats);

        // Call OpenAI API
        String aiResponse = callOpenAI(prompt);

        // Parse and return
        return parseAIResponse(aiResponse, stats);
    }

    private String buildPrompt(List<Task> tasks, UserStats stats) {
        StringBuilder prompt = new StringBuilder();
        prompt.append("You are a productivity coach. Analyze this week's data:\n\n");
        prompt.append("Tasks completed: ").append(tasks.stream()
                .filter(Task::getCompleted).count()).append("\n");
        prompt.append("Total tasks: ").append(tasks.size()).append("\n");
        prompt.append("XP earned: ").append(stats.getTotalXp()).append("\n");
        prompt.append("Current level: ").append(stats.getLevel()).append("\n");
        prompt.append("Streak: ").append(stats.getCurrentStreak()).append(" days\n\n");
        prompt.append("Provide:\n");
        prompt.append("1. Weekly summary (2-3 sentences)\n");
        prompt.append("2. Productivity tips (3 tips)\n");
        prompt.append("3. Areas for improvement\n");
        prompt.append("4. Motivation quote\n");
        prompt.append("Format as JSON with keys: summary, tips[], improvements, quote");

        return prompt.toString();
    }

    private String callOpenAI(String prompt) {
        // Use Spring WebClient or RestTemplate
        RestTemplate restTemplate = new RestTemplate();

        HttpHeaders headers = new HttpHeaders();
        headers.setBearerAuth(apiKey);
        headers.setContentType(MediaType.APPLICATION_JSON);

        Map<String, Object> body = Map.of(
            "model", "gpt-3.5-turbo",
            "messages", List.of(Map.of("role", "user", "content", prompt)),
            "temperature", 0.7
        );

        HttpEntity<Map<String, Object>> request = new HttpEntity<>(body, headers);

        ResponseEntity<Map> response = restTemplate.postForEntity(
                "https://api.openai.com/v1/chat/completions",
                request,
                Map.class);

        Map<String, Object> choices = (Map<String, Object>)
                ((List) response.getBody().get("choices")).get(0);
        Map<String, String> message = (Map<String, String>) choices.get("message");

        return message.get("content");
    }

    private AIReportDTO parseAIResponse(String response, UserStats stats) {
        try {
            ObjectMapper mapper = new ObjectMapper();
            JsonNode root = mapper.readTree(response);

            return AIReportDTO.builder()
                    .summary(root.get("summary").asText())
                    .tips(Arrays.asList(mapper.treeToValue(root.get("tips"), String[].class)))
                    .improvements(root.get("improvements").asText())
                    .quote(root.get("quote").asText())
                    .weekStart(LocalDate.now().with(DayOfWeek.MONDAY))
                    .weekEnd(LocalDate.now())
                    .build();
        } catch (Exception e) {
            log.error("Failed to parse AI response", e);
            return getFallbackReport();
        }
    }

    private AIReportDTO getFallbackReport() {
        return AIReportDTO.builder()
                .summary("Great work this week! Keep pushing forward.")
                .tips(Arrays.asList(
                    "Break large tasks into smaller ones",
                    "Use the 2-minute rule for quick tasks",
                    "Review your goals weekly"))
                .improvements("Try to maintain a consistent daily streak")
                .quote("The secret of getting ahead is getting started. - Mark Twain")
                .build();
    }
}
